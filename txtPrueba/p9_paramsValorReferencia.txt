program FuncionesParametrosPorValorYReferencia;

{ Función que retorna un valor entero }
function SumarValores(var resultado: integer): integer;
begin
  resultado := 2 + 4;  { El parámetro resultado es pasado por referencia y se modifica }
  SumarValores := resultado;  { La función retorna el resultado }
end;

var
  numero1, numero2, suma: integer;
  cadenaFinal: string;
  caracter: char;
begin
  { Asignación de valores }
  numero1 := 5;
  numero2 := 10;
  caracter := 'A';
  cadenaFinal := 'Resultado: ';
  suma := 0; { Inicializar la variable suma }

  { Imprimir las variables iniciales }
  writeln('Valor de numero1: ', numero1);
  writeln('Valor de numero2: ', numero2);
  writeln('Valor de caracter: ', caracter);
  writeln('Valor inicial de cadenaFinal: ', cadenaFinal);

  { Llamada a la función SumarValores pasando suma por referencia }
  writeln('Llamando a la función SumarValores...');
  writeln('Suma calculada: ', SumarValores(suma));

  { Imprimir el resultado después de la función }
  writeln('Valor de suma después de la función: ', suma);
end.
